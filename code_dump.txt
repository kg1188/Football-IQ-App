    - pixel_points: [[x1,y1],...,[x4,y4]]
    - yard_points: [[x1,y1],...,[x4,y4]] (yard coordinates, e.g., 0‑120, 0‑53.33)
    Returns the homography matrix as a list of lists.
    """
    global _homography_matrix
    pixel_pts = np.array(payload["pixel_points"], dtype="float32")
    yard_pts = np.array(payload["yard_points"], dtype="float32")
    H, status = cv2.findHomography(pixel_pts, yard_pts)
    _homography_matrix = H.tolist() if H is not None else None
    return {"homography": (_homography_matrix.tolist() if hasattr(_homography_matrix, "tolist") else _homography_matrix), "status": status.flatten().tolist() if hasattr(status, "tolist") else status}

import cv2
from ultralytics import YOLO

def process_video(input_video_path, output_video_path):
    """Process a video, track players with YOLOv8+ByteTrack, extract foot coordinates,
    and cluster tracks into teams based on torso colour.
    Returns a list of dicts with frame, id, foot_x, foot_y and team label.
    """
    # Load YOLO model (ByteTrack) – will use .track() with custom params later
    from sklearn.cluster import KMeans
    import numpy as np
    model = YOLO("yolov8n.pt")
    cap = cv2.VideoCapture(input_video_path)
    
    fps = int(cap.get(cv2.CAP_PROP_FPS))
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    
    fourcc = cv2.VideoWriter_fourcc(*"mp4v")
    out = cv2.VideoWriter(output_video_path, fourcc, fps, (width, height))
    
    frame_count = 0
    tracking_data = []
    # Store per‑track colour samples (RGB) for first 15 frames
    colour_samples = {}
    max_colour_frames = 15
    
    while cap.isOpened():
        success, frame = cap.read()
        if not success:
            break
        frame_count += 1
        # Run tracking with BYTETRACK, keep only class 0 (person), confidence 0.35
        results = model.track(frame, persist=True, tracker="bytetrack.yaml", classes=[0], conf=0.35)
        for result in results:
            for box in result.boxes:
                # Bounding box in xywh format
                x_center, y_center, w, h = box.xywh[0].tolist()
                track_id = int(box.id[0]) if box.id is not None else -1
                # Foot coordinate: bottom centre of the bounding box
                foot_x = x_center
                foot_y = y_center + h / 2
                tracking_data.append({
                    "frame": frame_count,
                    "id": track_id,
                    "foot_x": foot_x,
                    "foot_y": foot_y
                })
                # Colour clustering – extract torso region (25‑55% height, 30‑70% width of bbox)
                if frame_count <= max_colour_frames:
                    x1 = int(x_center - w / 2)
                    y1 = int(y_center - h / 2)
                    # Compute region bounds relative to bbox
                    torso_x0 = int(x1 + 0.30 * w)
                    torso_y0 = int(y1 + 0.25 * h)
                    torso_x1 = int(x1 + 0.70 * w)
                    torso_y1 = int(y1 + 0.55 * h)
                    torso = frame[torso_y0:torso_y1, torso_x0:torso_x1]
                    if torso.size > 0:
                        median_rgb = np.median(torso.reshape(-1, 3), axis=0)
                        colour_samples.setdefault(track_id, []).append(median_rgb)
        # Annotated frame for optional output video
        annotated_frame = results[0].plot()
        out.write(annotated_frame)
    cap.release()
    out.release()
    # -------------------------------------------------
    # Team colour clustering (K‑Means, 3 clusters)
    # -------------------------------------------------
    # Build a feature vector per track: median of its collected colours
    track_ids = []
    colour_vectors = []
    for tid, samples in colour_samples.items():
        if samples:
            median_vec = np.median(np.stack(samples), axis=0)
            track_ids.append(tid)
            colour_vectors.append(median_vec)
    if colour_vectors:
        kmeans = KMeans(n_clusters=3, random_state=0).fit(colour_vectors)
        label_map = {tid: int(lbl) for tid, lbl in zip(track_ids, kmeans.labels_)}
    else:
        label_map = {}
    # Attach team label to tracking data
    for entry in tracking_data:
        entry["team"] = label_map.get(entry["id"], -1)  # -1 = unknown / official
    return tracking_data

    # Load YOLO model with BYTETRACK tracker and confidence threshold
    model = YOLO("yolov8n.pt")
    # The tracking call will be customized in the processing loop

    cap = cv2.VideoCapture(input_video_path)
    
    fps = int(cap.get(cv2.CAP_PROP_FPS))
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    
    fourcc = cv2.VideoWriter_fourcc(*"mp4v")
    out = cv2.VideoWriter(output_video_path, fourcc, fps, (width, height))

    frame_count = 0
    tracking_data = []

    while cap.isOpened():
        success, frame = cap.read()
        if not success:
            break
            
        frame_count += 1
        results = model.track(frame, persist=True, classes=[0])
        
        for result in results:
            for box in result.boxes:
                x_center, y_center, w, h = box.xywh[0].tolist()
                track_id = int(box.id[0]) if box.id is not None else -1
                
                tracking_data.append({
                    "frame": frame_count,
                    "id": track_id,
                    "x": x_center,
                    "y": y_center
                })

        annotated_frame = results[0].plot()
        out.write(annotated_frame)

    cap.release()
    out.release()
    print(f"Processing complete! Saved to {output_video_path}")
    return tracking_data

import React, { useRef, useState } from "react";

/**
 * Calibration component â€“ Phase 3.
 * Shows the first frame of the uploaded video (passed via props) and lets the user
 * click four field landmarks. The pixel coordinates are sent to the backend
 * `/calibrate-homography/` endpoint together with a preset yard grid.
 */
export default function Calibration({ frame1Url, onCalibrated }) {
  const imgRef = useRef(null);
  const [points, setPoints] = useState([]); // [{x, y}, ...]
  const [status, setStatus] = useState(null);

  const handleClick = (e) => {
    if (!imgRef.current) return;
    const rect = imgRef.current.getBoundingClientRect();
    const x = Math.round(e.clientX - rect.left);
    const y = Math.round(e.clientY - rect.top);
    setPoints((prev) => {
      const next = [...prev, { x, y }];
      return next.slice(0, 4); // keep only the first four clicks
    });
  };

  const presetYardPoints = {
    highschool: [
      [0, 0],
      [120, 0],
      [120, 53.33],
      [0, 53.33],
    ],
    college: [
      [0, 0],
      [120, 0],
      [120, 53.33],
      [0, 53.33],
    ],
    nfl: [
      [0, 0],
      [120, 0],
      [120, 53.33],
      [0, 53.33],
    ],
  };

  const sendCalibration = async (yardKey) => {
    if (points.length !== 4) {
      setStatus("Please click exactly four points on the image.");
      return;
    }
    const payload = {
      pixel_points: points.map((p) => [p.x, p.y]),
      yard_points: presetYardPoints[yardKey],
    };
    try {
      const resp = await fetch("http://localhost:8000/calibrate-homography/", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const data = await resp.json();
      setStatus("Calibration completed.");
      onCalibrated(data.homography);
    } catch (err) {
      console.error(err);
      setStatus("Calibration failed.");
    }
  };

  const reset = () => {
    setPoints([]);
    setStatus(null);
  };

  const autoEstimate = () => {
    // Dummy autoâ€‘estimate â€“ just send empty payload (backend can generate a default)
    sendCalibration("highschool");
  };

  return (
    <div className="p-4 border rounded bg-gray-50">
      <h2 className="text-xl font-semibold mb-2">4â€‘Point Field Calibration</h2>
      <div className="relative inline-block">
        <img
          src={frame1Url}
          alt="First frame"
          ref={imgRef}
          onClick={handleClick}
          className="cursor-crosshair max-w-full border"
        />
        {/* render click markers */}
        {points.map((p, i) => (
          <div
            key={i}
            className="absolute w-2 h-2 bg-red-500 rounded-full transform -translate-x-1/2 -translate-y-1/2"
            style={{ left: `${p.x}px`, top: `${p.y}px` }}
          >
            <span className="sr-only">Point {i + 1}</span>
          </div>
        ))}
      </div>
      <div className="mt-4 space-x-2">
        <button
          className="px-3 py-1 bg-blue-600 text-white rounded"
          onClick={() => sendCalibration("highschool")}
        >
          Highâ€‘School
        </button>
        <button
          className="px-3 py-1 bg-green-600 text-white rounded"
          onClick={() => sendCalibration("college")}
        >
          College
        </button>
        <button
          className="px-3 py-1 bg-purple-600 text-white rounded"
          onClick={() => sendCalibration("nfl")}
        >
          NFL
        </button>
        <button
          className="px-3 py-1 bg-gray-600 text-white rounded"
          onClick={autoEstimate}
        >
          Skip / Autoâ€‘Estimate
        </button>
        <button
          className="px-3 py-1 bg-yellow-600 text-white rounded"
          onClick={reset}
        >
          Reset
        </button>
      </div>
      {status && <p className="mt-2 text-sm text-gray-700">{status}</p>}
    </div>
  );
}

===== SplitScreen.jsx =====
import React, { useEffect, useRef, useState } from "react";
import PropTypes from "prop-types";

/**
 * SplitScreen component â€“ Phase 4.
 * Left side: video element (controls provided by browser).
 * Right side: canvas rendering the football field and player dots synchronized to video time.
 * Props:
 *   videoUrl â€“ URL to MP4 (served via FastAPI static /films/)
 *   trackingData â€“ array of {frame, id, foot_x, foot_y, team}
 *   fps â€“ framesâ€‘perâ€‘second of the source video
 *   homography â€“ optional 3Ã—3 matrix (array of arrays) to map foot coordinates to yard coordinates.
 */
export default function SplitScreen({ videoUrl, trackingData, fps, homography }) {
  const videoRef = useRef(null);
  const canvasRef = useRef(null);
  const [canvasSize, setCanvasSize] = useState({ width: 800, height: 360 }); // default field view size

  // Preâ€‘process tracking data into a map: frame -> array of players
  const frameMap = useRef({});
  useEffect(() => {
    const map = {};
    trackingData.forEach((item) => {
      const f = item.frame;
      if (!map[f]) map[f] = [];
      map[f].push(item);
    });
    frameMap.current = map;
  }, [trackingData]);

  // Convert foot pixel coordinates to yard coordinates using homography (if provided)
  const applyHomography = (x, y) => {
    if (!homography) return { x, y };
    const [a, b, c] = homography[0];
    const [d, e, f] = homography[1];
    const [g, h, i] = homography[2];
    const denom = g * x + h * y + i;
    const X = (a * x + b * y + c) / denom;
    const Y = (d * x + e * y + f) / denom;
    return { x: X, y: Y };
  };

  // Draw field (simple green rectangle with yard lines) and players for a given frameIdx
  const drawFrame = (frameIdx) => {
    const ctx = canvasRef.current?.getContext("2d");
    if (!ctx) return;
    const { width, height } = canvasSize;
    ctx.clearRect(0, 0, width, height);

    // --- Draw field background ---
    ctx.fillStyle = "#006400"; // dark green
    ctx.fillRect(0, 0, width, height);
    ctx.strokeStyle = "white";
    ctx.lineWidth = 2;
    // yard lines every 10 yards (0â€‘120)
    for (let yard = 0; yard <= 120; yard += 10) {
      const x = (yard / 120) * width;
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, height);
      ctx.stroke();
    }
    // end zones (5â€‘yard zones on each side)
    ctx.fillStyle = "#003300";
    ctx.fillRect(0, 0, (5 / 120) * width, height);
    ctx.fillRect(width - (5 / 120) * width, 0, (5 / 120) * width, height);

    // --- Draw players ---
    const players = frameMap.current[frameIdx] || [];
    players.forEach((p) => {
      const { foot_x, foot_y } = p;
      const pos = applyHomography(foot_x, foot_y);
      // Map yard coordinates (0â€‘120, 0â€‘53.33) to canvas dimensions
      const cx = (pos.x / 120) * width;
      const cy = (pos.y / 53.33) * height;
      const radius = 6;
      const colour = p.team === 0 ? "#ff0000" : p.team === 1 ? "#0000ff" : "#aaaaaa"; // team A red, B blue, other gray
      ctx.beginPath();
      ctx.arc(cx, cy, radius, 0, Math.PI * 2);
      ctx.fillStyle = colour;
      ctx.fill();
      ctx.strokeStyle = "black";
      ctx.stroke();
    });
  };

  // Animation loop â€“ sync to video currentTime
  useEffect(() => {
    let animationId = null;
    const render = () => {
      if (videoRef.current && canvasRef.current) {
        const currentTime = videoRef.current.currentTime;
        const frameIdx = Math.max(1, Math.round(currentTime * fps));
        drawFrame(frameIdx);
      }
      animationId = requestAnimationFrame(render);
    };
    render();
    return () => cancelAnimationFrame(animationId);
  }, [fps, homography, canvasSize]);

  // Resize canvas to match container size when component mounts
  useEffect(() => {
    const resize = () => {
      if (canvasRef.current) {
        const parent = canvasRef.current.parentElement;
        if (parent) {
          const rect = parent.getBoundingClientRect();
          setCanvasSize({ width: rect.width, height: rect.height });
        }
      }
    };
    resize();
    window.addEventListener("resize", resize);
    return () => window.removeEventListener("resize", resize);
  }, []);

  return (
    <div className="grid grid-cols-2 gap-4">
      <div className="flex justify-center items-center">
        <video
          ref={videoRef}
          src={videoUrl}
          controls
          className="max-w-full h-auto"
          crossOrigin="anonymous"
        />
      </div>
      <div className="relative">
        <canvas ref={canvasRef} width={canvasSize.width} height={canvasSize.height} className="border" />
      </div>
    </div>
  );
}

SplitScreen.propTypes = {
  videoUrl: PropTypes.string.isRequired,
  trackingData: PropTypes.array.isRequired,
  fps: PropTypes.number.isRequired,
  homography: PropTypes.array,
};

 * Main application flow:
 * 1️⃣ Upload video (handled by backend /process-video/ endpoint).

    setHomography(matrix);
    setStatus('Calibration complete. Syncing video and field.');
  };

  return (
    <div className="min-h-screen bg-gray-900 text-white p-6 font-sans">
      <div className="max-w-4xl mx-auto bg-gray-800 p-6 rounded-lg shadow-lg border border-gray-700">
        <h1 className="text-3xl font-bold mb-4">Football Spatial Hub</h1>
        <p className="text-gray-400 mb-4">
          Upload an All‑22 MP4 clip to generate tactical maps and coaching insights.
        </p>
        {/* File selector */}
        <input
          type="file"
          accept="video/mp4"
          onChange={(e) => setFile(e.target.files[0])}
          className="block w-full text-sm text-gray-400 file:mr-4 file:py-2 file:px-4 file:rounded file:border-0 file:text-sm file:font-semibold file:bg-blue-600 file:text-white hover:file:bg-blue-700 mb-4"
        />
        <button
          onClick={handleProcess}
          className="bg-green-600 hover:bg-green-700 text-white font-bold py-2 px-6 rounded transition-colors"
        >
          Process Film
        </button>
        {status && <p className="mt-4 text-yellow-400 font-mono">{status}</p>}

        {/* Phase 3 – Calibration UI */}
        {videoInfo && !homography && (
          <Calibration frame1Url={videoInfo.frame1_url} onCalibrated={handleCalibrated} />
        )}

        {/* Phase 4 – Split‑screen video & field */}
        {videoInfo && homography && (
          <SplitScreen
            videoUrl={videoInfo.video_url}
            trackingData={videoInfo.tracking}
            fps={videoInfo.fps}
            homography={homography}
          />
        )}

        {/* Phase 5 – Coaching & scouting panels */}
        {videoInfo && homography && (
          <Phase5 trackingData={videoInfo.tracking} fps={videoInfo.fps} />
        )}
      </div>
    </div>
  );
}
